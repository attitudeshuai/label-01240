**核心功能实现：**

1. **选股策略模块** (`backend/app/services/strategy_service.py`)
   - 竞价选股：涨幅2%-5%、量比>1.5、流通市值>20亿
   - 三一选股：量比>2、涨幅>3%、换手率>3%
   - 缠论选股：超跌反弹、60日跌幅>10%
   - 龙头选股：涨停或接近涨停、成交额大
   - 趋势选股：5日涨幅>5%、10日涨幅>8%

   - 右下角浮动面板，可显示/隐藏
   - 自选股买卖点提醒

3. **热点板块** (`backend/app/services/sector_service.py`)
   - 行业板块/概念板块切换
   - 板块龙头股（龙一、龙二、龙三）

   - 盘口信息（今开、最高、最低、成交量等）

   - K线图（支持12种周期：5秒、15秒、1/5/15/30/60/120分钟、日/周/月线）

   - user: 用户表

   - sector: 板块信息

   - strategy_result: 策略结果
   - alert: 提醒消息
   - operation_log: 操作日志

**数据库建表语句** (`backend/schema.sql`)

```sql
-- 创建数据库
CREATE DATABASE IF NOT EXISTS stock_strategy DEFAULT CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;

USE stock_strategy;

-- 用户表
CREATE TABLE IF NOT EXISTS `user` (
    `id` INT AUTO_INCREMENT PRIMARY KEY,
    `username` VARCHAR(50) NOT NULL UNIQUE,
    `password_hash` VARCHAR(255) NOT NULL,
    `email` VARCHAR(100),
    `is_active` BOOLEAN DEFAULT TRUE,
    `created_at` DATETIME DEFAULT CURRENT_TIMESTAMP,
    `updated_at` DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- 股票基础信息表
CREATE TABLE IF NOT EXISTS `stock` (
    `code` VARCHAR(10) PRIMARY KEY,
    `name` VARCHAR(50) NOT NULL,
    `market` VARCHAR(10) NOT NULL,
    `sector_code` VARCHAR(20),
    `is_active` BOOLEAN DEFAULT TRUE,
    `list_date` DATE,
    `created_at` DATETIME DEFAULT CURRENT_TIMESTAMP
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- 板块信息表
CREATE TABLE IF NOT EXISTS `sector` (
    `code` VARCHAR(20) PRIMARY KEY,
    `name` VARCHAR(50) NOT NULL,
    `type` VARCHAR(20),
    `change_pct` DECIMAL(10,4) DEFAULT 0,
    `rank` INT DEFAULT 0,
    `updated_at` DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- 板块成分股关联表
CREATE TABLE IF NOT EXISTS `sector_stock` (
    `id` INT AUTO_INCREMENT PRIMARY KEY,
    `sector_code` VARCHAR(20) NOT NULL,
    `stock_code` VARCHAR(10) NOT NULL,
    `is_leader` BOOLEAN DEFAULT FALSE,
    `leader_rank` INT DEFAULT 0,
    `created_at` DATETIME DEFAULT CURRENT_TIMESTAMP,
    UNIQUE KEY `uk_sector_stock` (`sector_code`, `stock_code`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- 自选股分组表
CREATE TABLE IF NOT EXISTS `watchlist` (
    `id` INT AUTO_INCREMENT PRIMARY KEY,
    `user_id` INT NOT NULL,
    `name` VARCHAR(50) NOT NULL DEFAULT '默认分组',
    `sort_order` INT DEFAULT 0,
    `created_at` DATETIME DEFAULT CURRENT_TIMESTAMP
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- 自选股表
CREATE TABLE IF NOT EXISTS `watchlist_stock` (
    `id` INT AUTO_INCREMENT PRIMARY KEY,
    `watchlist_id` INT NOT NULL,
    `stock_code` VARCHAR(10) NOT NULL,
    `cost_price` DECIMAL(10,2) DEFAULT 0,
    `hold_volume` INT DEFAULT 0,
    `sort_order` INT DEFAULT 0,
    `created_at` DATETIME DEFAULT CURRENT_TIMESTAMP,
    UNIQUE KEY `uk_watchlist_stock` (`watchlist_id`, `stock_code`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- 策略配置表
CREATE TABLE IF NOT EXISTS `strategy_config` (
    `id` INT AUTO_INCREMENT PRIMARY KEY,
    `user_id` INT NOT NULL,
    `strategy_type` VARCHAR(30) NOT NULL,
    `name` VARCHAR(50) NOT NULL,
    `params` JSON,
    `is_enabled` BOOLEAN DEFAULT TRUE,
    `created_at` DATETIME DEFAULT CURRENT_TIMESTAMP,
    `updated_at` DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- 策略选股结果表
CREATE TABLE IF NOT EXISTS `strategy_result` (
    `id` INT AUTO_INCREMENT PRIMARY KEY,
    `strategy_type` VARCHAR(30) NOT NULL,
    `stock_code` VARCHAR(10) NOT NULL,
    `stock_name` VARCHAR(50),
    `signal_type` VARCHAR(20),
    `signal_price` DECIMAL(10,2),
    `signal_reason` VARCHAR(500),
    `score` INT DEFAULT 0,
    `trade_date` DATE NOT NULL,
    `created_at` DATETIME DEFAULT CURRENT_TIMESTAMP
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- 提醒消息表
CREATE TABLE IF NOT EXISTS `alert` (
    `id` INT AUTO_INCREMENT PRIMARY KEY,
    `user_id` INT NOT NULL,
    `stock_code` VARCHAR(10) NOT NULL,
    `stock_name` VARCHAR(50),
    `alert_type` VARCHAR(30) NOT NULL,
    `message` VARCHAR(500),
    `price` DECIMAL(10,2),
    `is_read` BOOLEAN DEFAULT FALSE,
    `created_at` DATETIME DEFAULT CURRENT_TIMESTAMP
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- 操作日志表
CREATE TABLE IF NOT EXISTS `operation_log` (
    `id` INT AUTO_INCREMENT PRIMARY KEY,
    `user_id` INT,
    `action` VARCHAR(50) NOT NULL,
    `module` VARCHAR(50),
    `detail` TEXT,
    `ip_address` VARCHAR(50),
    `created_at` DATETIME DEFAULT CURRENT_TIMESTAMP
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;
```

**Cursor**

```python
@router.get("/alerts")
async def get_watchlist_alerts(db: Session = Depends(get_db), current_user: User = Depends(get_current_user)):
    """获取自选股买卖点提醒 - 实时分析"""
    try:
        # 获取用户自选股
        watchlist = db.query(Watchlist).filter(Watchlist.user_id == current_user.id).first()
        if not watchlist:
            return ResponseModel(data=[])
        
        stocks = db.query(WatchlistStock).filter(WatchlistStock.watchlist_id == watchlist.id).all()
        if not stocks:
            return ResponseModel(data=[])
        
        # 获取实时行情并分析
        quote_service = get_quote_service(db)
        alerts = []
        
        for ws in stocks:
            quote = quote_service.get_realtime_quote(ws.stock_code)
            if not quote:
                continue
            
            # 买入信号：涨幅>3%且量比>1.5
            if quote.get('change_pct', 0) > 3 and quote.get('volume_ratio', 0) > 1.5:
                alerts.append({
                    "stock_code": ws.stock_code,
                    "stock_name": quote.get('name', ''),
                    "alert_type": "buy",
                    "message": f"放量上涨，涨幅{quote['change_pct']:.2f}%，量比{quote['volume_ratio']:.2f}",
                    "price": quote.get('price', 0)
                })
            # 卖出信号：跌幅>3%（止损）或涨幅>8%（止盈）
            elif quote.get('change_pct', 0) < -3:
                alerts.append({
                    "stock_code": ws.stock_code,
                    "stock_name": quote.get('name', ''),
                    "alert_type": "sell",
                    "message": f"止损提醒，跌幅{abs(quote['change_pct']):.2f}%",
                    "price": quote.get('price', 0)
                })
            elif quote.get('change_pct', 0) > 8:
                alerts.append({
                    "stock_code": ws.stock_code,
                    "stock_name": quote.get('name', ''),
                    "alert_type": "sell",
                    "message": f"止盈提醒，涨幅{quote['change_pct']:.2f}%",
                    "price": quote.get('price', 0)
                })
        
        return ResponseModel(data=alerts)
    except Exception as e:
        logger.error(f"获取自选股提醒失败: {str(e)}")
        return ResponseModel(code=500, message="获取提醒失败", data=[])
```
